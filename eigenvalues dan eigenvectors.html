def createqueue():
    q=[]
    return q

def dequeue(q,data):
    q.append(data)
    return q

def enqueue(q):
    data = q.pop(0)
    return data


def proses(list):
    a = createqueue()
    b = createqueue()
    tolak = createqueue()

    for nama in list:
        if len(a) < 3:
            dequeue(a,nama)

        elif len(b) < 3:
            dequeue(b,nama)

        else:
          dequeue(tolak,nama)


    print("kasir a : ")
    for nama in a:
        print(f"- {nama} ")

    print()

    print("kasir b : ")
    for nama in b:
        print(f"- {nama} ")

    print()

    print("pelnggan ditolak :")
    for nama in tolak:
       print(f"- {nama} ")

def proses(list):
    exspress = createqueue()
    reguler = createqueue()

    for nama,jumlah in list:
        if len(exspress) < 2:
            dequeue(exspress,(nama,jumlah))

        else:
            dequeue(reguler,(nama,jumlah))

    print("kasir exspress : ")
    waktu_exspress = 0
    for nama,jumlah in exspress:
        print(f"- {nama}")
        waktu_exspress = waktu_exspress + jumlah

    print()

    print("kasir reguler : ")
    waktu_reguler = 0
    for nama,jumlah in reguler:
        print(f"- {nama} ")
        waktu_reguler = waktu_reguler + jumlah
    print(f"total waktu layanan : {waktu_reguler} mnt")

def creatdequeue():
    d=[]
    return d

def addfront(d,data):
    d.insert(0,data)
    return d

def addrear(d,data):
    d.append(data)
    return d

def removefront(d):
    data = d.pop(0)
    return data

def removerear(d):
    data = d.pop()
    return data

def size(d):
    return len(d)



def proses(data):
    antrian = creatdequeue()
    for nama,kondisi in data:
        if kondisi == "kritis":
            addfront(antrian,(nama,kondisi))

        else:
            addrear(antrian,(nama,kondisi))



    print(f"urutan layanan : ")
    for nama,kondisi in antrian:
        if kondisi == "kritis":
          print(f"{nama}, {kondisi}")
        else:
          print(f"{nama}")
